cmake_minimum_required(VERSION 3.20...4.0)
get_filename_component(TOIT_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(PICO_SDK_PATH "${TOIT_ROOT}/third_party/pico-sdk" CACHE PATH "Pico SDK")
set(PICO_BOARD pico CACHE STRING "Board (pico, pico_w_2023 or pico_w)")
list(APPEND PICO_BOARD_HEADER_DIRS "${CMAKE_CURRENT_LIST_DIR}/boards")
list(REMOVE_DUPLICATES PICO_BOARD_HEADER_DIRS)
set(PICO_PLATFORM rp2040 CACHE STRING "Platform")
set(picotool_DIR "${TOIT_ROOT}/.cache/rp2350/install/lib/cmake/picotool" CACHE PATH "Pinned picotool")
find_package(picotool 2.3.1 EXACT CONFIG REQUIRED)
include("${PICO_SDK_PATH}/pico_sdk_init.cmake")
project(toit_rp2040 C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 11)
pico_sdk_init()
if (NOT PICO_RP2040 OR NOT PICO_BOARD MATCHES "^(pico|pico_w|pico_w_2023)$")
  message(FATAL_ERROR "This flash layout supports the 2 MiB Pico, Pico W and Pico W-2023.")
endif()

option(TOIT_RP2040_BUILD_VM "Build the Toit VM application instead of native recovery tools" OFF)
if (TOIT_RP2040_BUILD_VM)
  include(vm.cmake)
  return()
endif()

option(TOIT_RP2040_RECOVERY_TEST_APP "Build recovery protocol at the application address for hardware tests" OFF)
set(native_targets toit-rp2040-recovery toit-rp2040-probe)
if (TOIT_RP2040_RECOVERY_TEST_APP)
  list(APPEND native_targets toit-rp2040-recovery-test)
endif()
foreach(target ${native_targets})
  if (target STREQUAL "toit-rp2040-recovery" OR target STREQUAL "toit-rp2040-recovery-test")
    add_executable(${target} recovery.c update.c recovery-stdio.c)
  else()
    add_executable(${target} probe.c)
  endif()
  if (target STREQUAL "toit-rp2040-recovery")
    set(flash_start 0x10000000)
    set(flash_length 0x10000)
  else()
    set(flash_start 0x10020000)
    set(flash_length 0xf0000)
  endif()
  if (target STREQUAL "toit-rp2040-recovery-test")
    target_compile_definitions(${target} PRIVATE TOIT_RP2040_RECOVERY_TEST_APP=1)
  endif()
  file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/${target}-link")
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/${target}-link/memory_flash.incl"
    "MEMORY { FLASH(rx) : ORIGIN = ${flash_start}, LENGTH = ${flash_length} }\n")
  pico_add_linker_script_override_path(${target} "${CMAKE_CURRENT_BINARY_DIR}/${target}-link")
  target_link_libraries(${target} PRIVATE pico_stdlib pico_unique_id hardware_flash hardware_watchdog)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
  target_compile_definitions(${target} PRIVATE
    PICO_STACK_SIZE=0x1000
    PICO_STDIO_USB_STDOUT_TIMEOUT_US=1000
    PICO_STDIO_USB_ENABLE_RESET_VIA_BAUD_RATE=0
    PICO_STDIO_USB_ENABLE_RESET_VIA_VENDOR_INTERFACE=0)
  pico_enable_stdio_usb(${target} 1)
  pico_enable_stdio_uart(${target} 0)
  pico_set_program_name(${target} ${target})
  pico_add_extra_outputs(${target})
endforeach()
set(TOIT_RP2040_PROBE_VERSION 1 CACHE STRING "Native OTA probe version")
target_compile_definitions(toit-rp2040-probe PRIVATE TOIT_RP2040_PROBE_VERSION=${TOIT_RP2040_PROBE_VERSION})
if (PICO_BOARD STREQUAL "pico_w_2023")
  target_sources(toit-rp2040-probe PRIVATE modem-probe.c)
  target_link_libraries(toit-rp2040-probe PRIVATE hardware_uart)
endif()

option(TOIT_RP2040_PROBE_WIFI "Enable native radio/DHCP/UDP bring-up in the probe" OFF)
if (TOIT_RP2040_PROBE_WIFI)
  if (NOT PICO_BOARD STREQUAL "pico_w")
    message(FATAL_ERROR "The CYW43 probe requires the official pico_w board; W-2023 uses ESP8285 UART.")
  endif()
  target_sources(toit-rp2040-probe PRIVATE wifi-probe.c)
  target_include_directories(toit-rp2040-probe PRIVATE "${CMAKE_CURRENT_LIST_DIR}/include")
  target_compile_definitions(toit-rp2040-probe PRIVATE TOIT_RP2040_PROBE_WIFI=1 CYW43_PIO_CLOCK_DIV_DYNAMIC=1)
  target_link_options(toit-rp2040-probe PRIVATE -Wl,--wrap=read_reg_u32_swap)
  target_link_libraries(toit-rp2040-probe PRIVATE pico_cyw43_arch_lwip_poll)
endif()
